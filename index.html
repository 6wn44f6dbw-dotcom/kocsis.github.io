Google E-Tablo Entegrasyon Ayarları

Tüm Kullanıcıların E-Tabloya Kaydolması İçin Apps Script Kodu:

Google E-Tablonuzda Uzantılar > Apps Script bölümüne bu kodu yapıştırıp **Yeni Dağıtım (Web Uygulaması - Herkes)** olarak yayınlayın:

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  var sheetLogs = ss.getSheetByName("Kayıtlar") || ss.insertSheet("Kayıtlar");
  var sheetTeachers = ss.getSheetByName("Öğretmenler") || ss.insertSheet("Öğretmenler");
  var sheetStudents = ss.getSheetByName("Öğrenciler") || ss.insertSheet("Öğrenciler");
  
  return ContentService.createTextOutput(JSON.stringify({
    records: getSheetData(sheetLogs),
    teachers: getSheetData(sheetTeachers),
    students: getSheetData(sheetStudents)
  })).setMimeType(ContentService.MimeType.JSON);
}

function getSheetData(sheet) {
  var rows = sheet.getDataRange().getValues();
  if (rows.length < 2) return [];
  var headers = rows[0];
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = rows[i][j];
    }
    result.push(obj);
  }
  return result;
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var contents = JSON.parse(e.postData.contents);
    var action = contents.action;
    
    if (action === "addRecord") {
      var sheet = ss.getSheetByName("Kayıtlar") || ss.insertSheet("Kayıtlar");
      if (sheet.getLastRow() === 0) sheet.appendRow(["id", "date", "student", "subject", "topic", "correct", "incorrect", "blank", "total"]);
      var d = contents.data;
      sheet.appendRow([d.id, d.date, d.student, d.subject, d.topic, d.correct, d.incorrect, d.blank, d.total]);
    } 
    else if (action === "deleteRecord") {
      var sheet = ss.getSheetByName("Kayıtlar");
      if (sheet) {
        var data = sheet.getDataRange().getValues();
        for (var i = data.length - 1; i >= 1; i--) {
          if (String(data[i][0]) === String(contents.id)) {
            sheet.deleteRow(i + 1);
            break;
          }
        }
      }
    }
    else if (action === "syncUsers") {
      if (contents.teachers) {
        var sheetT = ss.getSheetByName("Öğretmenler") || ss.insertSheet("Öğretmenler");
        sheetT.clear();
        sheetT.appendRow(["id", "name", "subject", "code"]);
        contents.teachers.forEach(function(t) { sheetT.appendRow([t.id, t.name, t.subject, t.code]); });
      }
      if (contents.students) {
        var sheetS = ss.getSheetByName("Öğrenciler") || ss.insertSheet("Öğrenciler");
        sheetS.clear();
        sheetS.appendRow(["id", "name", "class", "code", "teacherId", "note", "weeklyGoal"]);
        contents.students.forEach(function(s) { sheetS.appendRow([s.id, s.name, s.class || "", s.code, s.teacherId || "", s.note || "", s.weeklyGoal || 0]); });
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
